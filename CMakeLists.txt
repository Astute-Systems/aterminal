cmake_minimum_required(VERSION 3.22)
file(STRINGS VERSION ATERMINAL_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS VERSION)
# VERSION holds SemVer (e.g. 1.0.0-rc.4); project() only accepts the numeric part.
string(REGEX REPLACE "^([0-9]+\\.[0-9]+\\.[0-9]+).*" "\\1" ATERMINAL_VERSION_NUMERIC "${ATERMINAL_VERSION}")
project(aterminal VERSION ${ATERMINAL_VERSION_NUMERIC} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(SDL2 REQUIRED)
find_package(Threads REQUIRED)
add_compile_options("$<$<CONFIG:Release>:-fstack-protector-strong;-D_FORTIFY_SOURCE=2>")
add_library(softrender STATIC third_party/softrender/src/parallel.cpp third_party/softrender/src/png.cpp
    third_party/softrender/src/post.cpp third_party/softrender/src/text.cpp)
add_library(softrender::softrender ALIAS softrender)
target_include_directories(softrender PUBLIC third_party/softrender/include)
target_link_libraries(softrender PUBLIC Threads::Threads)
add_library(adlib STATIC third_party/adlib/src/adlib_synth.cpp third_party/adlib/src/adlib_design.cpp)
target_include_directories(adlib PUBLIC third_party/adlib/include)
add_library(terminal STATIC src/terminal.cpp)
target_include_directories(terminal PUBLIC src)
target_compile_options(terminal PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(parser_test tests/parser_test.cpp)
target_link_libraries(parser_test PRIVATE terminal)
add_test(NAME parser COMMAND parser_test)
add_executable(fuzz_test tests/fuzz_test.cpp)
target_link_libraries(fuzz_test PRIVATE terminal)
add_test(NAME fuzz COMMAND fuzz_test)
add_library(pty_session STATIC src/session.cpp src/prompt.cpp)
target_link_libraries(pty_session PUBLIC terminal util)
target_compile_options(pty_session PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pty_test tests/pty_test.cpp)
target_link_libraries(pty_test PRIVATE pty_session)
add_test(NAME pty COMMAND pty_test)
set_tests_properties(pty PROPERTIES TIMEOUT 15)
add_library(display STATIC src/display.cpp)
target_link_libraries(display PUBLIC terminal softrender::softrender)
target_compile_options(display PRIVATE -Wall -Wextra -Wpedantic)
add_executable(display_test tests/display_test.cpp)
target_link_libraries(display_test PRIVATE display)
add_test(NAME display COMMAND display_test "${CMAKE_CURRENT_SOURCE_DIR}/assets/unifont.hex")
add_library(motd STATIC src/motd.cpp)
target_include_directories(motd PUBLIC src)
target_compile_options(motd PRIVATE -Wall -Wextra -Wpedantic)
add_executable(keys_test tests/keys_test.cpp)
target_include_directories(keys_test PRIVATE src)
target_link_libraries(keys_test PRIVATE terminal SDL2::SDL2)
add_test(NAME keys COMMAND keys_test)
add_executable(motd_test tests/motd_test.cpp)
target_link_libraries(motd_test PRIVATE motd terminal)
add_test(NAME motd COMMAND motd_test)
add_executable(aterminal src/main.cpp)
target_link_libraries(aterminal PRIVATE pty_session display motd adlib SDL2::SDL2)
target_compile_options(aterminal PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(aterminal PRIVATE ATERMINAL_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
add_test(NAME sdl_smoke COMMAND aterminal --self-test --no-sound --screenshot "${CMAKE_CURRENT_BINARY_DIR}/green.png")
add_test(NAME sdl_amber_audio COMMAND aterminal --self-test --amber --sound --screenshot "${CMAKE_CURRENT_BINARY_DIR}/amber.png")
add_test(NAME sdl_size_colour COMMAND aterminal --self-test --no-sound --size 800x600 --text-size 1 --colour green --fullscreen=false)
add_test(NAME sdl_fullscreen_flags COMMAND aterminal --self-test --no-sound --width=960 --height=640 --text-size=5 --color=amber --fullscreen=true)
set_tests_properties(sdl_smoke sdl_amber_audio sdl_size_colour sdl_fullscreen_flags PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy;SOFTRENDER_THREADS=2" TIMEOUT 30)
add_test(NAME flags_invalid_size COMMAND aterminal --size=0x600)
add_test(NAME flags_invalid_colour COMMAND aterminal --colour=purple)
add_test(NAME flags_invalid_text_size COMMAND aterminal --text-size=2abc)
add_test(NAME flags_invalid_fullscreen COMMAND aterminal --fullscreen=maybe)
add_test(NAME flags_invalid_text_size_high COMMAND aterminal --text-size=7)
add_test(NAME flags_startup_with_command COMMAND aterminal -c true -e true)
add_test(NAME flags_invalid_geometry COMMAND aterminal --geometry=80)
add_test(NAME flags_geometry_too_large COMMAND aterminal --geometry=500x200 --text-size=6)
add_test(NAME flags_geometry_with_size COMMAND aterminal --geometry=80x25 --size=800x600)
add_test(NAME flags_invalid_cursor COMMAND aterminal --cursor=round)
add_test(NAME flags_invalid_bell COMMAND aterminal --bell=loud)
add_test(NAME flags_invalid_console COMMAND aterminal --console=13)
add_test(NAME flags_invalid_scanlines COMMAND aterminal --scanlines=101)
add_test(NAME flags_invalid_directory COMMAND aterminal -d /nonexistent)
add_test(NAME flags_invalid_prompt COMMAND aterminal --prompt=nonsense)
add_test(NAME flags_prompt_needs_bash COMMAND aterminal --shell /bin/sh --prompt=dos)
add_test(NAME flags_exec_with_command COMMAND aterminal -e true -- /bin/true)
add_test(NAME flags_exec_empty COMMAND aterminal --)
add_test(NAME flags_missing_config COMMAND aterminal --config /nonexistent)
add_test(NAME flags_prompt_list COMMAND aterminal --prompt=list)
set_tests_properties(flags_prompt_list PROPERTIES PASS_REGULAR_EXPRESSION "slackware")
add_test(NAME sdl_options COMMAND aterminal --self-test --no-sound --geometry 100x30 --text-size 1 --cursor bar --no-blink --scrollback 500 --bell none --scanlines 50 --bloom 25 --title Retro --console 3 --maximized --hold)
add_test(NAME sdl_colour_crt COMMAND aterminal --self-test --no-sound --colour=colour)
add_test(NAME sdl_exit_quits COMMAND aterminal --no-sound -e "exit 0")
add_test(NAME sdl_config COMMAND aterminal --self-test --no-sound --config "${CMAKE_CURRENT_SOURCE_DIR}/tests/config_test.conf")
set_tests_properties(sdl_options sdl_colour_crt sdl_exit_quits sdl_config PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy;SOFTRENDER_THREADS=2" TIMEOUT 30)
add_test(NAME flags_invalid_motd COMMAND aterminal --motd=nonsense)
add_test(NAME flags_motd_demo COMMAND aterminal --motd --demo)
add_test(NAME flags_motd_list COMMAND aterminal --motd=list)
set_tests_properties(flags_motd_list PROPERTIES PASS_REGULAR_EXPRESSION "haiku")
set_tests_properties(flags_invalid_size flags_invalid_colour flags_invalid_text_size flags_invalid_text_size_high flags_invalid_fullscreen flags_startup_with_command flags_invalid_geometry flags_geometry_too_large flags_geometry_with_size flags_invalid_cursor flags_invalid_bell flags_invalid_console flags_invalid_scanlines flags_invalid_directory flags_invalid_prompt flags_prompt_needs_bash flags_exec_with_command flags_exec_empty flags_missing_config flags_invalid_motd flags_motd_demo PROPERTIES WILL_FAIL TRUE)
include(GNUInstallDirs)
install(TARGETS aterminal RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/unifont.hex assets/UNIFONT-LICENSE.txt assets/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/aterminal)
install(FILES third_party/softrender/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/aterminal RENAME SOFTRENDER-LICENSE.txt)
install(FILES third_party/adlib/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/aterminal RENAME ADLIB-LICENSE.txt)
configure_file(docs/aterminal.1.in aterminal.1 @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/aterminal.1" DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
install(FILES assets/aterminal.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES images/astute-terminal/Astute_Terminal_Icon.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME aterminal.svg)
install(FILES images/astute-terminal/Astute_Terminal_Icon_256.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps RENAME aterminal.png)
install(FILES images/astute-terminal/Astute_Terminal_Icon_512.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps RENAME aterminal.png)
install(FILES images/astute-terminal/Astute_Terminal_Icon_256.png DESTINATION ${CMAKE_INSTALL_DOCDIR}/images/astute-terminal)
install(FILES README.md CHANGELOG.md LICENSE THIRD-PARTY.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES assets/UNIFONT-LICENSE.txt DESTINATION ${CMAKE_INSTALL_DOCDIR}/assets)
install(PROGRAMS docs/colour-bars.sh docs/sample-session.sh DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
install(DIRECTORY docs/screenshots DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)

# Packages: ./install.sh --deb or --rpm. Arch Linux uses packaging/arch/PKGBUILD.
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME aterminal)
set(CPACK_PACKAGE_VERSION "${ATERMINAL_VERSION}")
# dpkg reads 1.0.0-rc.4 as upstream 1.0.0 with revision rc.4, which sorts above 1.0.0.
# Debian pre-releases use ~ (1.0.0~rc.4); only the deb Version and file name change.
string(REGEX REPLACE "^([0-9]+\\.[0-9]+\\.[0-9]+)-" "\\1~" CPACK_DEBIAN_PACKAGE_VERSION "${ATERMINAL_VERSION}")
set(CPACK_PACKAGE_VENDOR "Astute Systems")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Phosphor CRT terminal emulator")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_STRIP_FILES ON)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Ross Newman <ross.newman@astutesys.com>")
set(CPACK_DEBIAN_PACKAGE_SECTION x11)
set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://github.com/Astute-Systems/aterminal")
set(CPACK_PACKAGE_DESCRIPTION "A native Linux terminal with green or amber text, soft bloom, scanlines, a vignette, and optional AdLib-style key sounds. Twelve independent shell consoles live behind F1-F12.")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
set(CPACK_RPM_PACKAGE_RELEASE 1)
set(CPACK_RPM_PACKAGE_RELEASE_DIST ON)
set(CPACK_RPM_PACKAGE_URL "https://github.com/Astute-Systems/aterminal")
set(CPACK_RPM_PACKAGE_LICENSE "MIT")
set(CPACK_RPM_PACKAGE_GROUP "User Interface/X")
# Directories owned by the system's filesystem packages.
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /usr/share/applications /usr/share/doc /usr/share/icons
    /usr/share/icons/hicolor /usr/share/icons/hicolor/256x256 /usr/share/icons/hicolor/256x256/apps
    /usr/share/icons/hicolor/512x512 /usr/share/icons/hicolor/512x512/apps /usr/share/icons/hicolor/scalable
    /usr/share/icons/hicolor/scalable/apps /usr/share/man /usr/share/man/man1)
include(CPack)
